function pickFallingItemType(level: GameLevel, roll: number): FallingItemType {
  let checkpoint = 0;
  for (const [type, weight] of SPAWN_WEIGHTS[level]) {
    checkpoint += weight / 100;
    if (roll < checkpoint) return type;
  }
  return "record";
}
